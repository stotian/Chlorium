cmake_minimum_required(VERSION 3.25)
project(Chlorium VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build configuration" FORCE)
endif()
option(CHLORIUM_ENABLE_PROFILER "Build intrusive developer profiling hooks" OFF)
option(CHLORIUM_RUNTIME_AUDIT "Build autonomous in-game regression harness (never distribute)" OFF)
option(CHLORIUM_AUDIT_HARNESS_ONLY "Build only the native control harness" OFF)
option(CHLORIUM_ORBIT_BENCHMARK "Use the Orbit performance harness" OFF)
option(CHLORIUM_EXPERIMENTAL_STATIC_RENDERER "Build the incomplete Bismuth GPU replacement (development only)" OFF)
option(CHLORIUM_EXPERIMENTAL_NATIVE_MOVES "Test native movement loop dispatch optimization" OFF)
option(CHLORIUM_MEASUREMENT_ONLY "Measure an absent or separately packaged production mod" OFF)
option(CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING "Experimental native VBO streaming backend" OFF)
option(CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS "Experimental exact Area quotient reuse" OFF)
option(CHLORIUM_EXPERIMENTAL_MATRIX_REUSE "Experimental helper matrix linear-part reuse" OFF)
option(CHLORIUM_NATIVE_ATLAS_CACHE "Retired atlas upload cache, audit builds only" OFF)
if(CHLORIUM_NATIVE_ATLAS_CACHE AND NOT CHLORIUM_RUNTIME_AUDIT)
    message(FATAL_ERROR "The retired GPU atlas cache is restricted to audit builds")
endif()
if(CHLORIUM_NATIVE_ATLAS_CACHE AND (CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR
   CHLORIUM_ENABLE_PROFILER OR CHLORIUM_MEASUREMENT_ONLY OR CHLORIUM_AUDIT_HARNESS_ONLY OR
   CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING OR CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS OR
   CHLORIUM_EXPERIMENTAL_MATRIX_REUSE))
    message(FATAL_ERROR "Retired atlas cache audits must remain separate from other renderers")
endif()
if(CHLORIUM_EXPERIMENTAL_MATRIX_REUSE AND (CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS OR CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING OR CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR CHLORIUM_ENABLE_PROFILER OR CHLORIUM_MEASUREMENT_ONLY OR CHLORIUM_EXPERIMENTAL_NATIVE_MOVES OR CHLORIUM_AUDIT_HARNESS_ONLY))
    message(FATAL_ERROR "Matrix reuse must be isolated from other experiments")
endif()
if(CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS AND (CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING OR CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR CHLORIUM_ENABLE_PROFILER OR CHLORIUM_MEASUREMENT_ONLY OR CHLORIUM_EXPERIMENTAL_NATIVE_MOVES OR CHLORIUM_AUDIT_HARNESS_ONLY))
    message(FATAL_ERROR "Area arithmetic must be isolated from other experiments")
endif()
if(CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING AND (CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR CHLORIUM_ENABLE_PROFILER OR CHLORIUM_MEASUREMENT_ONLY OR CHLORIUM_EXPERIMENTAL_NATIVE_MOVES OR CHLORIUM_AUDIT_HARNESS_ONLY))
    message(FATAL_ERROR "Streaming must be isolated from other renderer experiments")
endif()
if(NOT CHLORIUM_RUNTIME_AUDIT)
    set(CHLORIUM_EXPERIMENTAL_NATIVE_MOVES OFF CACHE BOOL "Development only" FORCE)
    set(CHLORIUM_MEASUREMENT_ONLY OFF CACHE BOOL "Development only" FORCE)
endif()
if(CHLORIUM_MEASUREMENT_ONLY AND (NOT CHLORIUM_ORBIT_BENCHMARK OR CHLORIUM_EXPERIMENTAL_NATIVE_MOVES OR CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR CHLORIUM_ENABLE_PROFILER OR CHLORIUM_AUDIT_HARNESS_ONLY))
    message(FATAL_ERROR "Measurement-only requires the auto harness without production or experimental hooks")
endif()
if((CHLORIUM_AUDIT_HARNESS_ONLY OR CHLORIUM_ORBIT_BENCHMARK) AND NOT CHLORIUM_RUNTIME_AUDIT)
    message(FATAL_ERROR "Development harnesses require CHLORIUM_RUNTIME_AUDIT=ON")
endif()
if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    # Normalize French output before Ninja parses /showIncludes dependencies.
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
    set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/tools/msvc_launcher.py")
    set(CMAKE_C_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/tools/msvc_launcher.py")
endif()
include(FetchContent)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 0af55ccecd98d4e5a8d1fad7de25ba429d60e863)
FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG vulkan-sdk-1.4.335)
FetchContent_MakeAvailable(glm glslang)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
# Abandoned sorter and scratch entrypoint have no runtime role.
list(FILTER SOURCES EXCLUDE REGEX "/unused/|/culling/|/main\\.cpp$|/DifferenceMode\\.cpp$|/OverdrawView\\.cpp$")
if(NOT CHLORIUM_EXPERIMENTAL_STATIC_RENDERER)
    # Do not register inactive replacement-renderer hooks on native gameplay.
    list(FILTER SOURCES EXCLUDE REGEX "/renderer/hooks\\.cpp$")
endif()
if(NOT CHLORIUM_EXPERIMENTAL_STATIC_RENDERER AND NOT CHLORIUM_ENABLE_PROFILER)
    # The native path needs no shader compiler, static renderer or developer
    # scene hooks. Keep that implementation in sources without loading it.
    set(SOURCES src/renderer/NativeTransformCache.cpp src/BuildIdentity.cpp)
    if(CHLORIUM_NATIVE_ATLAS_CACHE)
        list(APPEND SOURCES src/renderer/NativeAtlasUploads.cpp)
    else()
        list(APPEND SOURCES src/renderer/NativeAtlasStatsEmpty.cpp)
    endif()
    if(CHLORIUM_RUNTIME_AUDIT)
        list(APPEND SOURCES src/BProfiler.cpp)
    endif()
endif()
if(CHLORIUM_AUDIT_HARNESS_ONLY)
    set(SOURCES tests/runtime_audit.cpp src/BuildIdentity.cpp)
endif()
if(CHLORIUM_MEASUREMENT_ONLY)
    set(SOURCES tests/NativeAtlasStatsStub.cpp src/BuildIdentity.cpp src/BProfiler.cpp)
endif()
if(CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING)
    set(SOURCES src/renderer/NativeStreamingUploads.cpp src/BuildIdentity.cpp)
    if(CHLORIUM_RUNTIME_AUDIT)
        list(APPEND SOURCES src/BProfiler.cpp)
    endif()
endif()
if(CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS)
    set(SOURCES src/renderer/NativeAreaDivisions.cpp src/renderer/NativeAtlasStatsEmpty.cpp src/BuildIdentity.cpp)
    if(CHLORIUM_RUNTIME_AUDIT)
        list(APPEND SOURCES src/BProfiler.cpp)
    endif()
endif()
if(CHLORIUM_EXPERIMENTAL_MATRIX_REUSE)
    set(SOURCES src/renderer/NativeMatrixReuse.cpp src/renderer/NativeAtlasStatsEmpty.cpp src/BuildIdentity.cpp)
    if(CHLORIUM_RUNTIME_AUDIT)
        list(APPEND SOURCES src/BProfiler.cpp)
    endif()
endif()
add_library(Chlorium SHARED ${SOURCES})
if(NOT CHLORIUM_EXPERIMENTAL_STATIC_RENDERER AND NOT CHLORIUM_ENABLE_PROFILER AND
   NOT CHLORIUM_NATIVE_ATLAS_CACHE AND NOT CHLORIUM_MEASUREMENT_ONLY AND
   NOT CHLORIUM_AUDIT_HARNESS_ONLY AND NOT CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING AND
   NOT CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS AND NOT CHLORIUM_EXPERIMENTAL_MATRIX_REUSE)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_NATIVE_GPU_TRANSFERS_ONLY=1)
endif()
if(CHLORIUM_EXPERIMENTAL_MATRIX_REUSE)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_EXPERIMENTAL_MATRIX_REUSE=1)
endif()
if(CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_EXPERIMENTAL_AREA_DIVISIONS=1)
endif()
if(CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_EXPERIMENTAL_NATIVE_STREAMING=1)
endif()
if(CHLORIUM_MEASUREMENT_ONLY)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_MEASUREMENT_ONLY=1)
endif()
if(CHLORIUM_EXPERIMENTAL_NATIVE_MOVES)
    if(NOT CHLORIUM_EXPERIMENTAL_STATIC_RENDERER AND NOT CHLORIUM_ENABLE_PROFILER)
        target_sources(Chlorium PRIVATE src/renderer/NativeObjectUpdates.cpp)
    endif()
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_EXPERIMENTAL_NATIVE_MOVES=1)
endif()
if(CHLORIUM_EXPERIMENTAL_STATIC_RENDERER)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_EXPERIMENTAL_STATIC_RENDERER=1)
endif()
if(CHLORIUM_AUDIT_HARNESS_ONLY)
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_AUDIT_HARNESS_ONLY=1)
endif()
if(CHLORIUM_RUNTIME_AUDIT)
    if(CHLORIUM_ORBIT_BENCHMARK)
        target_sources(Chlorium PRIVATE tests/auto_level_benchmark.cpp)
    else()
        target_sources(Chlorium PRIVATE tests/runtime_audit.cpp)
    endif()
    target_compile_definitions(Chlorium PRIVATE CHLORIUM_RUNTIME_AUDIT=1)
    target_link_libraries(Chlorium psapi)
endif()
target_include_directories(Chlorium PRIVATE src)
target_link_libraries(Chlorium glm::glm)
if(CHLORIUM_EXPERIMENTAL_STATIC_RENDERER OR CHLORIUM_ENABLE_PROFILER)
    target_link_libraries(Chlorium glslang::glslang glslang::glslang-default-resource-limits)
endif()
if(CHLORIUM_ENABLE_PROFILER)
    target_compile_definitions(Chlorium PRIVATE BISMUTH_ENABLE_PROFILER)
endif()
if(NOT GEODE_SDK)
    set(GEODE_SDK "$ENV{GEODE_SDK}" CACHE PATH "Geode SDK directory" FORCE)
endif()
if(NOT EXISTS "${GEODE_SDK}/CMakeLists.txt")
    message(FATAL_ERROR "Set GEODE_SDK to the Geode 5.11.2 SDK")
endif()
set(GEODE_DONT_INSTALL_MODS ON)
add_subdirectory("${GEODE_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/geode")
setup_geode_mod(Chlorium DONT_INSTALL)
if(GEODE_DISABLE_CLI_CALLS)
    if(CHLORIUM_RUNTIME_AUDIT)
        target_compile_definitions(Chlorium PRIVATE GEODE_MOD_ID="local.chlorium.audit")
    else()
        target_compile_definitions(Chlorium PRIVATE GEODE_MOD_ID="local.chlorium")
    endif()
endif()
set_target_properties(Chlorium PROPERTIES OUTPUT_NAME "local.chlorium")
if(MSVC)
    target_compile_options(Chlorium PRIVATE /utf-8 /EHsc /bigobj /W3)
    target_precompile_headers(GeodeBindings PRIVATE <Geode/Bindings.hpp>)
endif()
target_precompile_headers(Chlorium PRIVATE <Geode/Geode.hpp>)

add_executable(chlorium_gpu_checks tests/gpu_checks.cpp src/renderer/ShaderPreprocessor.cpp)
target_include_directories(chlorium_gpu_checks PRIVATE tests/fakes src
    "${GEODE_SDK}/loader/include/Geode/cocos/platform/third_party/win32/OGLES")
target_link_libraries(chlorium_gpu_checks PRIVATE glm::glm glslang::glslang
    glslang::glslang-default-resource-limits opengl32 user32 gdi32
    "${GEODE_SDK}/loader/include/link/win64/glew32.lib")
target_compile_options(chlorium_gpu_checks PRIVATE /EHsc /utf-8)

add_executable(chlorium_streaming_checks tests/streaming_gpu_checks.cpp)
target_include_directories(chlorium_streaming_checks PRIVATE src
    "${GEODE_SDK}/loader/include/Geode/cocos/platform/third_party/win32/OGLES")
target_link_libraries(chlorium_streaming_checks PRIVATE opengl32 user32 gdi32
    "${GEODE_SDK}/loader/include/link/win64/glew32.lib")
target_compile_options(chlorium_streaming_checks PRIVATE /EHsc /utf-8)
